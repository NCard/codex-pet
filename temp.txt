let isAlarmActive = false;
let snoozedAlarms = [];

function showAlarmBubble(alarm) {
  isAlarmActive = true;
  chatBubble.style.display = 'block';
  if (bubbleTimeout) clearTimeout(bubbleTimeout);
  
  const snoozeMins = alarm.snoozeInterval || 5;
  
  chatContent.innerHTML = `
    ${namePrefix}????嚗?br>
    <div style="margin: 5px 0; word-break: break-all;">${alarm.message}</div>
    <div style="display: flex; gap: 5px; margin-top: 8px;">
      <button id="btn-alarm-ok" style="flex:1; padding: 4px; border: none; background: #4caf50; color: white; border-radius: 4px; cursor: pointer; font-size: 12px; font-weight: bold;">???</button>
      <button id="btn-alarm-snooze" style="flex:1; padding: 4px; border: none; background: #ff9800; color: white; border-radius: 4px; cursor: pointer; font-size: 12px; font-weight: bold;">鞎芰 ${snoozeMins}??/button>
    </div>
  `;
  
  document.getElementById('btn-alarm-ok').onclick = () => {
    isAlarmActive = false;
    chatBubble.style.display = 'none';
  };
  
  document.getElementById('btn-alarm-snooze').onclick = () => {
    isAlarmActive = false;
    chatBubble.style.display = 'none';
    const triggerTime = Date.now() + snoozeMins * 60000;
    snoozedAlarms.push({ ...alarm, triggerTime });
  };
}
